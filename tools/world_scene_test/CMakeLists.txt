cmake_minimum_required(VERSION 3.20)
project(DarkWorldRendererTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(test_dds dds.cpp ../../libsrc/namedres/dds_decode.cpp)
add_test(NAME dds COMMAND test_dds)
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Vulkan REQUIRED)
set(RENDER_ROOT "${ENGINE_ROOT}/libsrc/lgd3d")
add_library(world_test_core STATIC
  "${RENDER_ROOT}/world_scene.cpp")
target_include_directories(world_test_core PUBLIC "${ENGINE_ROOT}/h")
foreach(pair "scene;main.cpp" "ranges;geometry_ranges.cpp" "history;history.cpp")
  list(GET pair 0 name)
  list(GET pair 1 source)
  add_executable(test_${name} "${source}")
  target_link_libraries(test_${name} PRIVATE world_test_core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_library(world_test_gpu STATIC
  "${RENDER_ROOT}/vulkan_world_features.cpp" "${RENDER_ROOT}/vulkan_world_geometry.cpp")
target_include_directories(world_test_gpu PRIVATE "${ENGINE_ROOT}/3rdparty/tracy/public")
target_link_libraries(world_test_gpu PUBLIC world_test_core Vulkan::Vulkan)
add_executable(test_geometry geometry.cpp)
target_link_libraries(test_geometry PRIVATE world_test_gpu)
add_test(NAME geometry COMMAND test_geometry)
set_tests_properties(geometry PROPERTIES SKIP_RETURN_CODE 77)
