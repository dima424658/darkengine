cmake_minimum_required(VERSION 3.20)
project(DarkWorldRendererTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(ENGINE_ROOT "../.." ABSOLUTE)
find_package(Vulkan REQUIRED)
if(NOT Vulkan_GLSLC_EXECUTABLE)
  message(FATAL_ERROR "Install the Vulkan SDK including glslc")
endif()
set(RENDER_ROOT "${ENGINE_ROOT}/libsrc/lgd3d")
add_library(world_test_core STATIC
  "${RENDER_ROOT}/world_scene.cpp" "${RENDER_ROOT}/world_shadows.cpp")
target_include_directories(world_test_core PUBLIC "${ENGINE_ROOT}/h")
foreach(pair "scene;main.cpp" "budget;shadow_budget.cpp" "cameras;shadow_cameras.cpp" "ranges;geometry_ranges.cpp")
  list(GET pair 0 name)
  list(GET pair 1 source)
  add_executable(test_${name} "${source}")
  target_link_libraries(test_${name} PRIVATE world_test_core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_library(world_test_gpu STATIC
  "${RENDER_ROOT}/vulkan_world_features.cpp" "${RENDER_ROOT}/vulkan_world_geometry.cpp"
  "${RENDER_ROOT}/vulkan_shadow_maps.cpp" "${RENDER_ROOT}/vulkan_world_renderer.cpp")
target_include_directories(world_test_gpu PRIVATE "${ENGINE_ROOT}/3rdparty/tracy/public")
target_link_libraries(world_test_gpu PUBLIC world_test_core Vulkan::Vulkan)
set(SHADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
file(MAKE_DIRECTORY "${SHADER_DIR}")
# Exercise the same optimization policy as the native renderer.
foreach(shader world.vert world_mapped.frag world_mapped_native.frag world_depth.frag world_composite.vert world_composite.frag shadow_map.vert shadow_map.frag)
  set(output "${SHADER_DIR}/${shader}.spv")
  add_custom_command(OUTPUT "${output}"
    COMMAND "${Vulkan_GLSLC_EXECUTABLE}" "$<IF:$<CONFIG:Debug>,-O0,-O>" --target-env=vulkan1.1 "${RENDER_ROOT}/shaders/${shader}" -o "${output}"
    DEPENDS "${RENDER_ROOT}/shaders/${shader}" "${RENDER_ROOT}/shaders/shadow_sampling.glsl" "${RENDER_ROOT}/shaders/world_mapped_body.glsl" VERBATIM)
  list(APPEND SHADERS "${output}")
endforeach()
add_custom_target(world_test_shaders DEPENDS ${SHADERS})
foreach(pair "geometry;geometry.cpp" "shadow_maps;shadow_maps.cpp" "lighting;lighting.cpp")
  list(GET pair 0 name)
  list(GET pair 1 source)
  add_executable(test_${name} "${source}")
  target_link_libraries(test_${name} PRIVATE world_test_gpu)
  add_dependencies(test_${name} world_test_shaders)
  if(name STREQUAL "geometry")
    add_test(NAME ${name} COMMAND test_${name})
  else()
    add_test(NAME ${name} COMMAND test_${name} "${SHADER_DIR}")
  endif()
  set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
